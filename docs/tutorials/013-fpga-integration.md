## Introduction

t27's Verilog backend enables FPGA synthesis directly from `.t27` specifications. This tutorial covers the complete workflow: from spec to bitstream, including deployment and testing on real hardware.

## Part 1: Why FPGA Integration?

- **Current reality:** No commercial ternary processors
- **t27 advantage:** Ready when ternary silicon arrives
- **FPGA bridge:** Prototype today on binary hardware

### Benefits

| Aspect | Software | FPGA |
|--------|----------|------|
| Latency | Variable | Deterministic |
| Throughput | CPU-limited | Parallel |
| Power | Higher | Lower |
| Reconfigurability | ❌ | ✅ |

### Use Cases

- **Neural network inference** — Low-latency GF operations
- **Signal processing** — Real-time φ-optimized filters
- **Cryptography** — Custom ternary primitives

## Part 3: FPGA-Ready Spec Writing

### Hardware-Friendly Constraints

For FPGA synthesis, specs must be:

1. **Combinational-friendly** — Avoid complex state machines in simple functions
2. **Fixed-size types** — No dynamic allocation
3. **Clock-aware** — Use `#clock` annotations for sequential logic
4. **Resource-aware** — Consider DSP block usage

### Basic FPGA Module

```t27
module FpgaGreeting {
    // Hardware-friendly constants
    const LED_COUNT: u8 = 8;
    const DELAY_CYCLES: u32 = 1000000;

    // Combinational logic
    fn parity_checker(data: u8) -> u1 {
        let mut parity: u1 = 0;
        for i in 0..8 {
            parity = parity xor ((data >> i) & 1);
        }
        return parity;
    }

    // Sequential logic (clocked)
    #clock
    fn blink_counter() -> u32 {
        // State persists across clock cycles
        static mut count: u32 = 0;
        count = count + 1;
        if count >= DELAY_CYCLES {
            count = 0;
        }
        return count;
    }

    // Test for verification
    test "parity works for all zeros" {
        given { }
        then { let result = parity_checker(0); }
        expect { result == 0; }
    }

    test "parity works for all ones" {
        given { }
        then { let result = parity_checker(0xFF); }
        expect { result == 0; }
    }

    test "parity works for single bit" {
        given { }
        then { let result = parity_checker(0x01); }
        expect { result == 1; }
    }
}
```

### GoldenFloat FPGA Module

```t27
module FpgaGoldenFloat {
    // GF16 to 16-bit wire representation
    struct GF16 {
        sign: u1,
        exponent: u6,
        mantissa: u9,
    }

    // Combinational GF16 addition
    fn gf16_add(a: GF16, b: GF16) -> GF16 {
        // Align exponents
        let exp_diff = (a.exponent as i8) - (b.exponent as i8);

        // Simplified: assume same exponent
        if exp_diff == 0 {
            let mant_sum = a.mantissa + b.mantissa;
            return GF16 {
                sign: a.sign,
                exponent: a.exponent,
                mantissa: mant_sum,
            };
        }

        // Handle different exponents
        // (Full implementation would denormalize)
        return a;
    }

    test "gf16_add same exponent" {
        given {
            let x = GF16 { sign: 0, exponent: 31, mantissa: 256 };
            let y = GF16 { sign: 0, exponent: 31, mantissa: 256 };
        }
        then {
            let result = gf16_add(x, y);
        }
        expect {
            result.mantissa == 512;
        }
    }
}
```

## Part 4: Generating Verilog

### Basic Generation

```bash
# Generate Verilog from spec
tri gen-verilog FpgaGreeting.t27 > gen/fpga_greeting.v
```

### Generated Verilog Structure

```verilog
// Generated by t27 compiler
// Source: FpgaGreeting.t27

`timescale 1ns / 1ps

module parity_checker (
    input  [7:0] data,
    output       parity
);
    // Combinational logic
    assign parity = ^data;
endmodule

module blink_counter (
    input  clk,
    input  rst,
    output [31:0] count
);
    // Sequential logic
    reg [31:0] count_reg;
    always @(posedge clk or posedge rst) begin
        if (rst) begin
            count_reg <= 32'd0;
        end else begin
            if (count_reg >= 32'd1000000) begin
                count_reg <= 32'd0;
            end else begin
                count_reg <= count_reg + 32'd1;
            end
        end
    end
    assign count = count_reg;
endmodule
```

### Top-Level Module Template

Create `rtl/top.v` for your FPGA:

```verilog
`timescale 1ns / 1ps

module t27_top (
    input  clk,
    input  rst_n,
    input  [7:0] switches,
    output [7:0] leds
);

    // Generated parity checker
    wire parity_bit;
    parity_checker parity_inst (
        .data(switches),
        .parity(parity_bit)
    );

    // Blink counter for status LED
    wire [31:0] blink_count;
    blink_counter blink_inst (
        .clk(clk),
        .rst(~rst_n),
        .count(blink_count)
    );

    // Output
    assign leds[0] = parity_bit;
    assign leds[1] = blink_count[23];  // Slow blink
    assign leds[7:2] = 6'b0;

endmodule
```

## Part 9: GoldenFloat FPGA Accelerator

### GF16 Multiply-Accumulate (MAC)

```t27
module GF16MAC {
    struct GF16 {
        sign: u1,
        exponent: u6,
        mantissa: u9,
    }

    // GF16 multiplication
    fn gf16_mul(a: GF16, b: GF16) -> GF16 {
        let sign_out = a.sign xor b.sign;
        let exp_out = a.exponent + b.exponent - 31;  // Adjust for bias
        let mant_out = (a.mantissa * b.mantissa) >> 9;

        return GF16 {
            sign: sign_out,
            exponent: exp_out,
            mantissa: mant_out,
        };
    }

    // GF16 addition
    fn gf16_add(a: GF16, b: GF16) -> GF16 {
        // Simplified: assume aligned
        let mant_sum = a.mantissa + b.mantissa;
        let exp_sum = a.exponent;
        let sign_sum = a.sign;  // Assume same sign

        return GF16 {
            sign: sign_sum,
            exponent: exp_sum,
            mantissa: mant_sum,
        };
    }

    // MAC operation: acc = acc + a * b
    #clock
    fn gf16_mac(acc: GF16, a: GF16, b: GF16) -> GF16 {
        let product = gf16_mul(a, b);
        let result = gf16_add(acc, product);
        return result;
    }

    // Test MAC
    test "mac computes correctly" {
        given {
            let acc = GF16 { sign: 0, exponent: 31, mantissa: 256 };
            let x = GF16 { sign: 0, exponent: 31, mantissa: 128 };
            let y = GF16 { sign: 0, exponent: 31, mantissa: 2 };
        }
        then {
            let result = gf16_mac(acc, x, y);
        }
        expect {
            result.mantissa == (256 + (128 * 2 >> 9));
        }
    }
}
```

### Pipeline Architecture

```verilog
// Pipelined GF16 MAC for high throughput
module gf16_mac_pipelined (
    input  clk,
    input  rst_n,
    input  [15:0] acc_in,
    input  [15:0] a_in,
    input  [15:0] b_in,
    input  valid_in,
    output [15:0] result,
    output       valid_out
);

    reg [15:0] stage1_acc;
    reg [15:0] stage1_a;
    reg [15:0] stage1_b;
    reg        stage1_valid;

    reg [15:0] stage2_product;
    reg [15:0] stage2_acc;
    reg        stage2_valid;

    // Stage 1: Multiplication
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            stage1_valid <= 1'b0;
        end else begin
            stage1_acc   <= acc_in;
            stage1_a     <= a_in;
            stage1_b     <= b_in;
            stage1_valid <= valid_in;
        end
    end

    // Stage 2: Accumulation
    assign stage2_product = gf16_mul(stage1_a, stage1_b);
    assign stage2_acc     = stage1_acc;

    // Stage 3: Output
    assign result    = gf16_add(stage2_acc, stage2_product);
    assign valid_out = stage2_valid;

endmodule
```

## Part 10: Resource Utilization and Timing

### Understanding FPGA Resources

| Resource | Purpose | Typical Usage |
|----------|---------|---------------|
| LUTs | Combinational logic | General computation |
| FFs | Sequential logic | Registers, state |
| DSP | Multipliers | GF16 MAC units |
| BRAM | Memory | Weight storage |
| MMCM/PLL | Clock management | Clock domains |

### Reading Utilization Reports

```bash

# Intel Quartus
```

### Typical GF16 MAC Resources (Artix-7)

| Component | LUTs | FFs | DSPs | BRAM |
|-----------|------|-----|------|------|
| GF16 Mul | 120 | 80 | 1 | 0 |
| GF16 Add | 85 | 60 | 0 | 0 |
| GF16 MAC | 205 | 140 | 1 | 0 |
| 16 MACs | 3280 | 2240 | 16 | 0 |

### Timing Closure

```tcl
# Generate timing report
report_timing_summary -file timing.txt

# Critical path analysis
report_timing -sort_by slack -max_paths 10 -file critical_paths.txt
```

If timing fails:
1. **Add pipeline stages** — Reduce combinatorial depth
2. **Reduce fanout** — Duplicate registers
3. **Use DSP slices** — For multiplication
4. **Optimize constraints** — Relax non-critical paths

## Part 13: FAQ

### Q: Can I use any FPGA?

A: Yes! t27 generates standard Verilog. Use vendor tools for synthesis.

### Q: How much FPGA resources does GF16 use?

A: A single GF16 MAC uses ~1 DSP slice, ~200 LUTs, and ~140 FFs.

### Q: Can I simulate before programming hardware?

A: Yes! Use Icarus Verilog, Verilator, or ModelSim for simulation.

### Q: What about ternary FPGAs?

A: No commercial ternary FPGAs exist yet. t27 is ready when they arrive.

### Q: How do I debug on hardware?

A: Use ILA (Integrated Logic Analyzer) in Vivado or SignalTap in Quartus.
